package frc.robot.commands;

import java.util.Optional;

import com.ctre.phoenix6.swerve.SwerveRequest;

import edu.wpi.first.apriltag.AprilTagFieldLayout;
import edu.wpi.first.apriltag.AprilTagFields;
import edu.wpi.first.math.controller.PIDController;
import edu.wpi.first.math.geometry.Pose3d;
import edu.wpi.first.math.geometry.Translation2d;
import edu.wpi.first.wpilibj2.command.Command;
import edu.wpi.first.wpilibj2.command.CommandScheduler;
import frc.robot.Constants;
import frc.robot.LimelightHelpers;
import frc.robot.RobotContainer;

public class MoveAndAlign extends Command{
    double t = 0;
    double goal;
    double Rotation;
    PIDController pid;
    SwerveRequest drive;
    AprilTagFieldLayout field = AprilTagFieldLayout.loadField(AprilTagFields.k2026RebuiltAndymark);
    boolean Stop = false;
    Optional<Pose3d> TagPose;
    Translation2d CurrentTranslation;
    Translation2d TargetVector;
    Translation2d HubTranslation;
    int ID;

    public MoveAndAlign(){
    }

    @Override
    public void initialize(){
        pid = new PIDController(0.115, 0, 0);
        pid.enableContinuousInput(-180, 180);
        //Stop = !Constants.MoveAndAlignEnabled;
    }

    @Override
    public void execute(){
        ID = (int) LimelightHelpers.getFiducialID(Constants.AprilTagLimelight1Constants.limelightName);
        TagPose = field.getTagPose(ID);
        //Getting the tag pose, storing it and checking whether an initial tag was found
        HubTranslation = new Translation2d(4.625594, 4.034536);

        CurrentTranslation = Constants.drivetrain.getState().Pose.getTranslation();
        Rotation = Constants.drivetrain.getState().Pose.getRotation().getDegrees();

        //System.out.println(Rotation);
        TargetVector = CurrentTranslation.minus(HubTranslation);
        goal = TargetVector.getAngle().getDegrees();
        //System.out.println(goal);
        //System.out.println(Rotation);

        double rotationalSpeed = pid.calculate(Rotation + 180, goal);
        //System.out.println(rotationalSpeed);

        drive = new SwerveRequest.FieldCentric()
        .withVelocityX(-RobotContainer.joystick.getLeftY() * RobotContainer.MaxSpeed)
        .withVelocityY(-RobotContainer.joystick.getLeftX() * RobotContainer.MaxSpeed)
        .withRotationalRate(rotationalSpeed); 

        CommandScheduler.getInstance().schedule(Constants.drivetrain.applyRequest(() -> drive).until(() -> RobotContainer.joystick.cross().getAsBoolean()));
  
    }  
    @Override
    public boolean isFinished(){
        return(Stop);
    }

    @Override
    public void end(boolean interrupted){
        System.out.println("End");
        pid.close();
        //CommandScheduler.getInstance().cancelAll();
    }
}
