package frc.robot.commands;

import java.util.Vector;

import edu.wpi.first.math.geometry.Pose2d;
import edu.wpi.first.math.geometry.Rotation2d;
import edu.wpi.first.math.geometry.Transform2d;
import edu.wpi.first.math.geometry.Translation2d;
import edu.wpi.first.wpilibj2.command.Command;
import frc.robot.Constants;
import frc.robot.LimelightHelpers;
import frc.robot.LimelightHelpers.RawDetection;

public class AutoIntake extends Command{
    
    private int GridHeight = 1;
    private int GridWidth = 1;
    private int GridMaxHeight = 10;
    private int GridMaxWidth = 10;

    private int[][] grid = new int[GridMaxWidth / GridWidth][GridMaxHeight / GridHeight]; 

    int ShiftAmount = GridMaxWidth - 2 * GridWidth;

    private int BestGridX = 0;
    private int BestGridY = 0;

    private Pose2d BestPose = new Pose2d();

    public AutoIntake(){

    }

    public int MaxElement(int[] list){
        int max = list[0];
        int place = 0;

        for (int i = 0; i < list.length; i++){
            if (list[i] > max){
                max = list[i];
                place = i;
            }
        }
        return place;
    }

    @Override
    public void execute(){
        int RowNumber = 0;
        
        RawDetection[] detection = LimelightHelpers.getRawDetections("");
        int[] ColumnsMax = new int[GridMaxWidth / GridMaxWidth];
        int[] RowsMaxPlace = new int[GridMaxHeight / GridHeight];

        for (RawDetection det : detection){
            int XCoordinate = Math.floorDiv((int) det.corner0_X, GridWidth) + ShiftAmount;
            int YCoordinate = Math.floorDiv((int) det.corner0_Y, GridHeight);

            grid[XCoordinate][YCoordinate] += 1;
        }

        for (int[] rows : grid){
            RowsMaxPlace[RowNumber] = MaxElement(rows);
            ColumnsMax[RowNumber] = rows[RowsMaxPlace[RowNumber]];
        }

        BestGridX = MaxElement(ColumnsMax);
        BestGridY = RowsMaxPlace[BestGridX];
        double Rotation = Math.atan((GridHeight * BestGridY) / ((GridWidth * BestGridX - ShiftAmount)));
        BestPose = Constants.drivetrain.getState().Pose.plus(new Transform2d(new Translation2d((double) (BestGridX * GridWidth - ShiftAmount), (double) BestGridY * GridHeight), new Rotation2d(Rotation)));
    }
}
