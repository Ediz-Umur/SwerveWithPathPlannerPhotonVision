    public void AlignLimelight(){
        double ID = LimelightHelpers.getFiducialID("");
        double XOffset = 0;
        if (ID == 21
         || ID == 26
         || ID == 20
         || ID == 18){
            XOffset = 0;
        } else 
        if (ID == 25
         || ID == 27
         || ID == 19
         || ID == 24){
            XOffset = 0.35;
            if (ID == 24){
                XOffset *= -1;
            }
         }
         LimelightHelpers.setFiducial3DOffset("", -0.5965, xOffset, -0.66);
    }

    public double calculateDistance(){
        xAngle = LimelightHelpers.getTX("");
        double yAngle = LimelightHelpers.getTY("");

        if (tagNumbers.size() < 0){
            return(0.0);
        }

        double tyDistance = (hubHeight - Constants.AprilTagLimelight1Constants.limelightHeight) / Math.tan(Units.degreesToRadians(Constants.AprilTagLimelight1Constants.limelightAngle + yAngle));
        double groundDistance = tyDistance / Math.cos(Units.degreesToRadians(xAngle));
        return(groundDistance);
    }
